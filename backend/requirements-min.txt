# Minimal deps to run the API with rule-based fallbacks.
# Heavy ML/LLM libs are optional; the app degrades gracefully without them.
#
# These are PINNED on purpose. Unpinned installs resolve to whatever is newest
# at deploy time, so a release of pandas or scikit-learn can break a build that
# worked yesterday without a single line of our code changing. Versions match
# requirements.txt so the two environments stay in step.

# Core API
fastapi==0.115.5
uvicorn[standard]==0.32.1
python-multipart==0.0.12
pydantic==2.10.3
PyYAML==6.0.2
httpx==0.28.1

# Data / DB
SQLAlchemy==2.0.36
psycopg2-binary==2.9.9
python-dotenv==1.0.1
pandas==2.2.3
numpy==2.1.3

# ML — scikit-learn must match the version the model bundles in models/ were
# trained with, or joblib refuses to load them and the API silently drops to
# its rule-based path. The build retrains against this same pin.
scikit-learn==1.5.2
joblib==1.4.2
